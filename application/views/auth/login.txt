<script>
    var currentRole = "siswa";
    var isJadwalAktifServer = <?= $is_jadwal_aktif ? 'true' : 'false'; ?>;

    function switchRole(e) {
        e.preventDefault();
        var title = $('#login-title');
        var desc = $('#login-desc');
        var mapel = $('#mapel-wrapper');
        var mapelSelect = $('#id_ujian');
        var submitBtn = $('#submit');
        var switchLink = $('#btn-switch-role');
        var alertBox = $('#db-alert-box');
        
        $('#infoMessage').html('');

        if (currentRole === "siswa") {
            currentRole = "admin";
            $('#role_akses').val('admin'); 
            
            title.text("Selamat Datang");
            desc.text("");
            
            mapel.hide();
            if(alertBox.length) { alertBox.hide(); } 
            
            mapelSelect.removeAttr('required').val('');
            submitBtn.addClass('admin-mode').text('Login');
            switchLink.html('<i class="fas fa-graduation-cap mr-1"></i> Login Siswa');
        } else {
            currentRole = "siswa";
            $('#role_akses').val('siswa'); 
            
            title.text("Selamat Datang");
            desc.text("");
            
            if (isJadwalAktifServer) {
                mapel.show();
                mapelSelect.attr('required', 'required');
            } else {
                mapel.hide();
                mapelSelect.removeAttr('required');
                if(alertBox.length) { alertBox.show(); } 
            }
            
            submitBtn.removeClass('admin-mode').text('Login Siswa');
            switchLink.html('<i class="fas fa-user-shield mr-1"></i> Login');
        }
    }

    $(document).ready(function(){
        $('#toggle-password').on('click', function () {
            const field = $('#password');
            const type = field.attr('type') === 'password' ? 'text' : 'password';
            field.attr('type', type);
            $(this).toggleClass('fa-eye fa-eye-slash');
        });

        $('form#login').on('submit', function(e){
            e.preventDefault(); 
            var infobox = $('#infoMessage');
            var btn = $('#submit');
            var textDefault = (currentRole === "siswa") ? "Login" : "Login";
            
            infobox.html('<div class="alert alert-info">Memeriksa data...</div>');
            btn.attr('disabled', 'disabled').text('Tunggu...');

            $.ajax({
                url: $(this).attr('action'),
                type: 'POST',
                data: $(this).serialize(),
                success: function(data){
                    btn.removeAttr('disabled').text(textDefault);
                    try {
                        var res = (typeof data === 'object') ? data : JSON.parse(data);
                        if(res.status){
                            infobox.html('<div class="alert alert-success">Login Berhasil</div>');
                            window.location.href = '<?= base_url() ?>' + res.url;
                        } else {
                            infobox.html('<div class="alert alert-danger">' + (res.failed ? res.failed : 'Username atau Password Salah!') + '</div>');
                        }
                    } catch (e) {
                        infobox.html('<div class="alert alert-warning">Terjadi kesalahan pada respon server.</div>');
                    }
                },
                error: function() {
                    btn.removeAttr('disabled').text(textDefault);
                    infobox.html('<div class="alert alert-danger">Gagal terhubung ke server. Cek koneksi Anda.</div>');
                }
            });
        });
    });
</script>
